const fs = require('fs');
const path = require('path');
const { execSync } = require('child_process');

const ROOT = process.cwd();
const readJSON = (p, fb) => { try { return JSON.parse(fs.readFileSync(p, 'utf8')); } catch { return fb; } };

function detectRepo() {
  if (process.env.GITHUB_REPOSITORY) return process.env.GITHUB_REPOSITORY.split('/');
  try {
    const url = execSync('git remote get-url origin', { stdio: ['ignore', 'pipe', 'ignore'] }).toString().trim();
    const m = url.match(/github\.com[:/]+([^/]+)\/([^/]+?)(?:\.git)?$/);
    if (m) return [m[1], m[2]];
  } catch { /* git nahi hai ya remote nahi */ }
  return ['USERNAME', path.basename(ROOT)];
}

async function gh(url) {
  const headers = { 'User-Agent': 'generate-readme', Accept: 'application/vnd.github+json' };
  if (process.env.GITHUB_TOKEN) headers.Authorization = 'Bearer ' + process.env.GITHUB_TOKEN;
  const res = await fetch('https://api.github.com' + url, { headers });
  if (!res.ok) throw new Error(url + ' -> ' + res.status);
  return res.json();
}

async function loadMeta(owner, repo) {
  if (process.env.MOCK_META) return readJSON(process.env.MOCK_META, { info: {}, languages: {} });
  try {
    const [info, languages] = await Promise.all([gh(`/repos/${owner}/${repo}`), gh(`/repos/${owner}/${repo}/languages`)]);
    return { info, languages };
  } catch (e) {
    console.warn('GitHub se info nahi mili (private repo ho to GITHUB_TOKEN set karein). Defaults use ho rahe hain:', e.message);
    return { info: {}, languages: {} };
  }
}

const LANG_ICONS = { JavaScript: 'js', TypeScript: 'ts', HTML: 'html', CSS: 'css', SCSS: 'sass', Python: 'py', Java: 'java', C: 'c', 'C++': 'cpp', 'C#': 'cs', PHP: 'php', Go: 'go', Rust: 'rust', Ruby: 'ruby', Kotlin: 'kotlin', Swift: 'swift', Dart: 'dart', Shell: 'bash', Vue: 'vue' };
const DEP_ICONS = { react: 'react', vue: 'vue', next: 'nextjs', express: 'express', vite: 'vite', tailwindcss: 'tailwind', bootstrap: 'bootstrap', typescript: 'ts', '@angular/core': 'angular', svelte: 'svelte', mongoose: 'mongodb', mongodb: 'mongodb' };

const titleCase = (s) => s.replace(/[-_]+/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase()).trim();
const banner = (s) => encodeURIComponent(s.replace(/[-_]+/g, ' ').slice(0, 80));
const badgeText = (s) => encodeURIComponent(s.replace(/-/g, '--').replace(/_/g, '__'));

function detectStack(languages, pkg, override) {
  if (override && override.length) return override;
  const ids = [];
  Object.keys(languages).forEach((l) => LANG_ICONS[l] && ids.push(LANG_ICONS[l]));
  if (pkg) {
    const deps = { ...(pkg.dependencies || {}), ...(pkg.devDependencies || {}) };
    Object.keys(deps).forEach((d) => DEP_ICONS[d] && ids.push(DEP_ICONS[d]));
    ids.push('nodejs');
  }
  ids.push('git', 'github');
  return [...new Set(ids)];
}

function detectInstall(pkg, override) {
  if (override && override.length) return override;
  const has = (f) => fs.existsSync(path.join(ROOT, f));
  if (pkg) {
    const s = pkg.scripts || {};
    return ['npm install', s.dev ? 'npm run dev' : 'npm start'];
  }
  if (has('requirements.txt')) return ['pip install -r requirements.txt', has('main.py') ? 'python main.py' : 'python app.py'];
  if (has('index.html')) return ['# No install needed, just open it in your browser', 'open index.html'];
  return [];
}

function tree() {
  const skip = new Set(['.git', 'node_modules', '.github', '.DS_Store', 'generate-readme.js', 'README.backup.md']);
  const entries = fs.readdirSync(ROOT, { withFileTypes: true })
    .filter((e) => !skip.has(e.name))
    .sort((a, b) => (b.isDirectory() - a.isDirectory()) || a.name.localeCompare(b.name))
    .slice(0, 20);
  return entries.map((e, i) => (i === entries.length - 1 ? '└──' : '├──') + ' ' + e.name + (e.isDirectory() ? '/' : ''));
}

async function main() {
  if (process.argv.includes('--action')) {
    const dir = path.join(ROOT, '.github', 'workflows');
    fs.mkdirSync(dir, { recursive: true });
    fs.writeFileSync(path.join(dir, 'readme.yml'), WORKFLOW);
    console.log('Created .github/workflows/readme.yml  (ab is file ko bhi commit karein: generate-readme.js)');
    return;
  }

  const [owner, repo] = detectRepo();
  const pkg = readJSON(path.join(ROOT, 'package.json'), null);
  const { info, languages } = await loadMeta(owner, repo);
  const C = CONFIG;
  const t = { gold: 'E5B33A', lightGold: 'F2C14E', bg: '0a0a0a', border: '3A2F12', text: 'F1ECE0', ...(C.theme || {}) };

  const topics = info.topics || [];
  const description = C.tagline || info.description || 'A project by ' + owner;
  const category = C.category || topics.slice(0, 3).join(' • ').toUpperCase();
  const demo = C.demo || info.homepage || '';
  const version = (C.version || (pkg && pkg.version && 'v' + pkg.version) || 'v1.0.0').replace(/-/g, '--');
  const tags = (C.tags && C.tags.length ? C.tags : topics).slice(0, 6);

  let featuresSection = '';
  const f = C.features || [];
  if (f.length) {
    const cell = (x) => (x ? `    <td width="50%">\n      <h3>${x.icon || '✨'} ${x.title}</h3>\n      ${x.text}\n    </td>` : '    <td width="50%"></td>');
    const rows = [];
    for (let i = 0; i < f.length; i += 2) rows.push(`  <tr>\n${cell(f[i])}\n${cell(f[i + 1])}\n  </tr>`);
    featuresSection = `---\n\n## ✨ Key Features\n\n<table>\n${rows.join('\n')}\n</table>\n\n`;
  }

  const install = detectInstall(pkg, C.install);
  const fence = '```';
  const quickstart = install.length
    ? `---\n\n## 🚀 Quick Start\n\n${fence}bash\n# Clone the repository\ngit clone https://github.com/${owner}/${repo}.git\ncd ${repo}\n\n${install.join('\n')}\n${fence}\n\n`
    : '';
  const lines = tree();
  const structure = lines.length ? `---\n\n## 📂 Project Structure\n\n${fence}text\n${repo}/\n${lines.join('\n')}\n${fence}\n\n` : '';

  const vars = {
    REPO_NAME: repo, OWNER: owner, TITLE: titleCase(repo), AUTHOR: C.author || owner,
    BANNER_TITLE: banner(titleCase(repo)), BANNER_DESC: banner(description),
    GOLD: t.gold, LIGHT_GOLD: t.lightGold, BG: t.bg, TEXT: t.text, BORDER: t.border, VERSION: version,
    CATEGORY_LINE_BLOCK: category ? `<sub><b>${category}</b></sub>\n\n<br/>\n` : '',
    LICENSE_BADGE: info.license && info.license.spdx_id && info.license.spdx_id !== 'NOASSERTION'
      ? `<img src="https://img.shields.io/github/license/${owner}/${repo}?style=for-the-badge&color=${t.gold}&labelColor=${t.bg}" />\n` : '',
    DEMO_BUTTON: demo
      ? `<a href="${demo}">\n  <img src="https://img.shields.io/badge/Live%20Demo-${t.bg}?style=for-the-badge&logo=googlechrome&logoColor=${t.lightGold}&labelColor=${t.bg}" />\n</a>\n` : '',
    ABOUT: C.about || description,
    SCREENSHOT: C.screenshot ? `<p align="center"><img src="${C.screenshot}" width="90%" /></p>\n` : '',
    FEATURES_SECTION: featuresSection,
    STACK_ICONS: detectStack(languages, pkg, C.stack).join(','),
    QUICKSTART_SECTION: quickstart,
    STRUCTURE_SECTION: structure,
    TAGS: tags.map((x) => `<img src="https://img.shields.io/badge/-%23${badgeText(x)}-${t.bg}?style=flat-square&labelColor=${t.bg}&color=${t.border}" />`).join('\n'),
  };

  const out = TEMPLATE.replace(/\{\{(\w+)\}\}/g, (m, k) => (k in vars ? vars[k] : m)).replace(/\n{3,}/g, '\n\n');

  const target = path.join(ROOT, 'README.md');
  if (fs.existsSync(target) && !fs.readFileSync(target, 'utf8').includes('generated by generate-readme.js')) {
    fs.copyFileSync(target, path.join(ROOT, 'README.backup.md'));
    console.log('Purani README.md ka backup: README.backup.md');
  }
  fs.writeFileSync(target, out);
  console.log(`README.md generated for ${owner}/${repo}`);
}

main();
